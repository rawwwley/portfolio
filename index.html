// Text scramble — resolves random chars to target string left-to-right.
// SCRAMBLE_DEPTH controls how many ticks a char stays noisy before locking.
const SCRAMBLE_CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_/.';
const TICK_MS        = 25;
const SCRAMBLE_DEPTH = 6;

function scrambleText(element, targetText) {
  if (scrambleInterval) clearInterval(scrambleInterval);
  let tick = 0;
  const totalTicks = SCRAMBLE_DEPTH + targetText.length;

  scrambleInterval = setInterval(() => {
    let out = '';
    for (let i = 0; i < targetText.length; i++) {
      if (targetText[i] === ' ') {
        out += ' ';
      } else if (i <= tick - SCRAMBLE_DEPTH) {
        out += targetText[i]; // locked — real char visible
      } else {
        out += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
      }
    }
    element.textContent = out;
    if (++tick > totalTicks) {
      clearInterval(scrambleInterval);
      scrambleInterval = null;
      element.textContent = targetText; // guarantee exact final string
    }
  }, TICK_MS);
}

module osc_capture #(
  parameter DATA_WIDTH = 12,
  parameter DEPTH      = 1024
)(
  input  wire                  clk,      // 100 MHz system clock
  input  wire                  rst_n,    // active-low reset
  input  wire [DATA_WIDTH-1:0] adc_data, // parallel ADC input
  input  wire                  trigger,  // rising-edge capture start
  output reg  [DATA_WIDTH-1:0] mem [0:DEPTH-1],
  output reg                   capture_done
);
  integer i;
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) capture_done <= 1'b0;
    else if (trigger) begin
      for (i = 0; i < DEPTH; i = i + 1)
        mem[i] <= adc_data;
      capture_done <= 1'b1;
    end
  end
endmodule

async function auditFolder(ctx, folderUrl, results = []) {
  const items = await ctx.web
    .getFolderByServerRelativeUrl(folderUrl)
    .files.select('Name', 'ServerRelativeUrl', 'TimeLastModified')();

  for (const file of items) {
    const issues = checkNamingConvention(file.Name);
    if (issues.length) {
      results.push({
        path:     file.ServerRelativeUrl,
        modified: file.TimeLastModified,
        issues
      });
    }
  }

  const subFolders = await ctx.web
    .getFolderByServerRelativeUrl(folderUrl)
    .folders.select('ServerRelativeUrl')();

  for (const folder of subFolders) {
    await auditFolder(ctx, folder.ServerRelativeUrl, results);
  }

  return results;
}

const RULES = [
  {
    id:   'no-spaces',
    test: name => !/ /.test(name),
    msg:  'Filename contains spaces — use underscores or hyphens'
  },
  {
    id:   'date-prefix',
    // Expects YYYY-MM-DD or YYYYMMDD at start of filename
    test: name => /^(\d{4}-\d{2}-\d{2}|\d{8})/.test(name),
    msg:  'Missing date prefix (expected YYYY-MM-DD)'
  },
  {
    id:   'no-special-chars',
    test: name => !/[#%&{}\\<>*?/$!'":@+`|=]/.test(name),
    msg:  'Contains characters invalid in SharePoint paths'
  }
];

export function checkNamingConvention(filename) {
  return RULES
    .filter(rule => !rule.test(filename))
    .map(rule => ({ rule: rule.id, message: rule.msg }));
}
